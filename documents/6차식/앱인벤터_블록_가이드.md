```
when Screen1.Initialize
do
  // 설정 불러오기
  if TinyDB1.GetValue("serverUrl") ≠ ""
  then
    set global serverUrl to TinyDB1.GetValue("serverUrl")
  
  if TinyDB1.GetValue("confidence") ≠ ""
  then
    set global confidenceThreshold to TinyDB1.GetValue("confidence")
  
  if TinyDB1.GetValue("captureInterval") ≠ ""
  then
    set global captureInterval to TinyDB1.GetValue("captureInterval")
  
  // UI 초기화
  set Label_Status.Text to "대기 중..."
  set Label_DetectionCount.Text to "탐지: 0회"
  set Label_LastDetection.Text to "마지막 탐지: 없음"

  // 서버 연결 테스트
  call CheckServerConnection
```

```
      set Label_Status.TextColor to Red

  then
    call ProcessDetectionResult(responseContent)

  then
    call ProcessLogsResult(responseContent)
  
  else if contains(url, "/stats")
  then
    call ProcessStatsResult(responseContent)
```

### 8. 이미지를 서버로 전송

```
procedure SendImageToServer(base64Image)
do
  // URL 설정
  set Web1.Url to join(global serverUrl, "/detect")
  
  // 헤더 설정
  set Web1.RequestHeaders to make a list
    make a list "Content-Type" "application/json"
  
  // JSON 데이터 생성
  set jsonData to join(
    "{",
    "\"image\":\"", base64Image, "\",",
    "\"confidence\":", global confidenceThreshold,
    "}"
  )
  
  // POST 요청
  call Web1.PostText(jsonData)
```

### 9. 탐지 결과 처리

```
procedure ProcessDetectionResult(jsonResponse)
do
  // JSON 파싱 (JsonTextDecode 사용)
  set responseDict to call JsonTextDecode(jsonResponse)
  
  // success 확인
  if get success from responseDict
  then
    // detections 배열 가져오기
    set detections to get detections from responseDict
    set detectionCount to length of detections
    
    if detectionCount > 0
    then
      // 탐지 횟수 증가
      set global detectionCount to (global detectionCount + 1)
      set Label_DetectionCount.Text to join("탐지: ", global detectionCount, "회")
      
      // 마지막 탐지 결과 저장
      set global lastDetections to detections
      
      // 바운딩 박스 그리기
      call DrawBoundingBoxes(detections)
      
      // 탐지 정보 업데이트
      call UpdateLastDetectionLabel(detections)
      
      // 알림
      call ShowNotification(detections)
      
      // 소리 알림
      if global enableSound
      then
        call Sound1.Play
      
      // 음성 알림
      if global enableVoice
      then
        call SpeakDetection(detections)
    else
      // 탐지된 객체 없음
      call Canvas1.Clear
      set Canvas1.BackgroundImage to global currentImage
  else
    // 오류 처리
    set errorMsg to get error from responseDict
    call Notifier1.ShowAlert(join("오류: ", errorMsg))
```

### 10. 바운딩 박스 그리기

```
procedure DrawBoundingBoxes(detections)
do
  // 캔버스 초기화 (배경 이미지 유지)
  // 이미지 위에 그리기 위해 Clear 대신 새로 그림
  
  // 각 탐지 결과에 대해 반복
  for each detection in detections
  do
    // bbox 정보 가져오기 (x, y, w, h)
    set bbox to get bbox from detection
    set x to get x from bbox
    set y to get y from bbox
    set w to get w from bbox
    set h to get h from bbox
    
    // 좌상단, 우하단 좌표 계산
    set x1 to x - (w / 2)
    set y1 to y - (h / 2)
    set x2 to x + (w / 2)
    set y2 to y + (h / 2)
    
    // 캔버스 크기에 맞게 스케일링
    set scaleX to Canvas1.Width / get width from responseDict.image_size
    set scaleY to Canvas1.Height / get height from responseDict.image_size
    
    set x1_scaled to x1 * scaleX
    set y1_scaled to y1 * scaleY
    set x2_scaled to x2 * scaleX
    set y2_scaled to y2 * scaleY
    
    // 바운딩 박스 그리기
    set Canvas1.PaintColor to Red
    set Canvas1.LineWidth to 3
    
    // 사각형 그리기 (4개의 선)
    call Canvas1.DrawLine(x1_scaled, y1_scaled, x2_scaled, y1_scaled)  // 상단
    call Canvas1.DrawLine(x2_scaled, y1_scaled, x2_scaled, y2_scaled)  // 우측
    call Canvas1.DrawLine(x2_scaled, y2_scaled, x1_scaled, y2_scaled)  // 하단
    call Canvas1.DrawLine(x1_scaled, y2_scaled, x1_scaled, y1_scaled)  // 좌측
    
    // 레이블 텍스트
    set className to get class from detection
    set confidence to get confidence from detection
    set labelText to join(className, " ", round(confidence * 100), "%")
    
    // 텍스트 그리기
    set Canvas1.FontSize to 14
    set Canvas1.PaintColor to Yellow
    call Canvas1.DrawText(labelText, x1_scaled, y1_scaled - 5)
```

### 11. 탐지 정보 업데이트

```
procedure UpdateLastDetectionLabel(detections)
do
  // 텍스트 초기화
  set detectionText to ""
  
  // 각 탐지 결과 추가
  for each detection in detections
  do
    set className to get class from detection
    set confidence to get confidence from detection
    set bbox to get bbox from detection
    
    set detectionText to join(
      detectionText,
      "• ", className, " (", round(confidence * 100), "%)", "\n",
      "  좌표: x=", round(get x from bbox), 
      ", y=", round(get y from bbox), "\n",
      "  크기: w=", round(get w from bbox), 
      ", h=", round(get h from bbox), "\n\n"
    )
  
  // 라벨 업데이트
  set Label_LastDetection.Text to detectionText
```

```
procedure ShowNotification(detections)
do

  // 알림 메시지
  set message to join(

    "신뢰도: ", round(confidence * 100), "%"

  // 알림 표시
  call Notifier1.ShowAlert(message)
```

## 확장 프로그램

### 필요한 확장 프로그램

1. **ImageToBase64** (이미지를 Base64로 변환)
   - 다운로드: MIT App Inventor Extensions Gallery
   - 또는 직접 제작

2. **JsonTextDecode** (JSON 파싱)
   - 앱인벤터 내장 블록 사용 가능
   - 또는 Web 컴포넌트의 JsonTextDecode 사용

### ImageToBase64 확장 프로그램 대체 방법

앱인벤터에서 직접 Base64 인코딩이 어려운 경우:

**방법 1: 이미지 파일 경로 전송**
- 이미지를 서버에 직접 업로드
- Multipart/form-data 사용

**방법 2: 중간 서버 사용**
- 이미지를 임시 서버에 업로드
- URL을 Flask 서버로 전송

## 문제 해결

### 이미지가 전송되지 않음
- ImageToBase64 확장 프로그램 확인
- 이미지 크기 확인 (너무 크면 실패)
- 네트워크 연결 확인

### 바운딩 박스가 잘못된 위치에 표시됨
- 스케일링 계산 확인
- Canvas 크기와 이미지 크기 비율 확인

### 앱이 느림
- 촬영 간격 늘리기 (2초 이상)
- 이미지 크기 줄이기
- 서버 성능 확인
